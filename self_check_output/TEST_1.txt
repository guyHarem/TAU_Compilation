
.data

string_illegal_div_by_0: .asciiz "Illegal Division By Zero"

global_g: .word 0

.text
.globl _start

_start:
	subu $sp, $sp, 4
	sw $ra, 0($sp)
	jal main
	lw $ra, 0($sp)
	addu $sp, $sp, 4
	li $v0, 10
	syscall

main:
	li $t0,1
	move $t2, $t0
	li $t0,0
	blt $t0,$t1,Label_2_AssignOne
	bge $t0,$t1,Label_3_AssignZero

Label_2_AssignOne:
	li $t0,1
	j Label_1_end

Label_3_AssignZero:
	li $t0,0
	j Label_1_end

Label_1_end:
	beq $t0,$zero,Label_0_end
	lw $t0,global_g
	add $t1,$t2,$t0
	li $t0,1
	add $t0,$t1,$t0
	move $t0, $t0
	li $t0,1
	add $t0,$t2,$t0
	move $t2, $t0

Label_0_end:
	jr $ra

label_print_int:
	move $t0, $a0
	move $a0,$t0
	li $v0,1
	syscall
	li $a0,32
	li $v0,11
	syscall
	jr $ra

label_print_string:
	move $t0, $a0

	addi $s1, $s1, 1

	move $v0, $t0
	jr $ra

	move $s1, $t1
strcopy_start:
	lb $s2, 0($s0)
	beq $s2, $zero, strcopy_end

	addi $s0, $s0, 1
	addi $s1, $s1, 1
	j strcopy_start
strcopy_end:
	sb $zero, 0($s1)
	jr $ra
